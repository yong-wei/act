MATLAB 程序：

% 建立开环传递函数模型

G = zpk($-2 -2$, $-10 -100 -0.75+2.38i -0.75-2.38i$, 1); z = 0.707;

% 绘制相应系统的根轨迹

figure(1) % 图(f)

rlocus(G); axis equal;

figure (2) % 图（d）

rlocus(G); sgrid(z,[]); hold on;      % 取阻尼比为0.707

axis($-6 2 -5 5$); K = 1430;          % 最佳阻尼比为对应的根轨迹增益

rlocus(G,K)                           % 最佳阻尼比为0.707时，系统的闭环特征根

figure (3)  % 图（e）

rlocus(G); axis equal; hold on; axis($-120 20 -150 150$);

K = 14300; rlocus(G,K)                % K=14300时,系统的闭环特征根

% 中重量巡航时的时间响应

K1 = 0.02; K2 = 71500; wn = 2.5; kos = 0.3;

num1 = $K2 4*K2 4*K2$; den1 = [1 110 1000];

num2 = [K1]; den2 = $1 2*wn * kos wn^2$;

[numc, denc] = series(num1, den1, num2, den2);

[num, den] = cloop(numc, denc); roots(den); % 系统闭环传递函数与闭环极点

sys = tf(num, den); t = 0:0.001:7;

figure(4) % 图（g）

step(sys,t); grid on;

% 轻重量降落时的时间响应

K1 = 0.2; K2 = 71500; wn=2.5; kos=0.3;

num1 = $K2 4*K2 4*K2$; den1 = [1 110 1000];

num2 = [K1]; den2 = $1 2*wn * kos wn^2$;

[numc, denc] = series(num1, den1, num2, den2);

[num, den] = cloop(numc, denc); roots(den); % 系统闭环传递函数与闭环极点

sys = tf(num, den); t = 0:0.001:3;

figure(5) % 图（h）

step(sys,t); grid on;
