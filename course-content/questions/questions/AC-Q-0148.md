T = 0:0.01:30;

% 初始条件 e(0) = 4, e(0) = 0

% 求解初始条件 e(0) = 4, e(0) = 0 下的系统微分方程

figure(1)

plot(t,e(:,1),e(:,2));grid

figure(2)

plot(t,e(:,1));grid

% 绘制系统的误差响应曲线

function de = sys805(t,e)

% 描述系统的微分方程

del = e(2);

if (e(1) < -2)

de2 = -2 - e(1);

else (abs(e(1)) < 2)

de2 = 0;

end

de = [del de2];
