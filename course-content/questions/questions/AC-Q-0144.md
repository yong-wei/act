K = [2.5,2.4,1];

T = 0.25;

t = 0:T:140;

u = 2 + t;

e = zeros(length(t),length(K));

for ii = 1:length(K)

G = tf(K(ii),[1,0],'inputdelay',0.5);

Gz = c2d(G,T);

Phie = feedback(1,Gz);

e(:,ii) = lsim(Phie,u,t,0);

end
