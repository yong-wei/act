plot(t,c(:,1));

set(findall(gcf,'type','line'),'linewidth',LineWidth);

function dc = sys(t,c)

dc1 = c(2);

if c(1)<-0.04||$(c(1)<0.04&&c(1)>-0.04)&&c(2)>0$

dc2 = 2-c(2);

else

end
