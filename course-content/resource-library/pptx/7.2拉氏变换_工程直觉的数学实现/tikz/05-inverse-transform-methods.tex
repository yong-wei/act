  signal/.style={->, line width=1.2pt, draw=black!80, >=Stealth}
}

\begin{document}
\begin{tikzpicture}[node distance=0.9cm]
  \node[box] (a) {proper rational\\function?};
  \node[box, right=of a] (b) {yes: partial fractions\\no repeated roots};
  \node[box, right=of b] (c) {repeated roots:\\higher-order terms};
  \node[box, right=of c] (d) {undetermined\\coefficients};
  \draw[signal] (a) -- (b);
  \draw[signal] (b) -- (c);
  \draw[signal] (c) -- (d);
