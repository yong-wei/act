      $s^2+2\zeta\omega_n s+\omega_n^2=0$
    \end{minipage}
  };
  \node[card, right=1.0cm of left] (right) {
    \begin{minipage}{5.0cm}

      $0<\zeta<1$: underdamped\\
      $\zeta=1$: critical damping\\
      $\zeta>1$: overdamped
    \end{minipage}
  };
